Start a campaign for a new CVE

curl -s -X POST {{ B }}/campaigns \
  -H "Authorization: Bearer $ZDRS_KEY" -H "Content-Type: application/json" \
  -d '{"cve_id":"CVE-2026-12345","product_pattern":"nginx","version_constraint":"< 1.27.3"}'

Close an issue from CI after deploying a fix

curl -s -X PATCH {{ B }}/issues/42 \
  -H "Authorization: Bearer $ZDRS_KEY" -H "Content-Type: application/json" \
  -d '{"status":"fixed","comment":"Patched in release 2026.11.1"}'

Pull open critical issues into a SIEM

curl -s "{{ B }}/issues?severity=critical&limit=500" \
  -H "Authorization: Bearer $ZDRS_KEY"

Scan and wait for the result

JOB=$(curl -s -X POST {{ B }}/scans/nuclei -H "Authorization: Bearer $ZDRS_KEY" \
  -H "Content-Type: application/json" \
  -d '{"targets":["https://app.example.com"],"mode":"groups","groups":["http/cves"],"severities":["critical","high"]}' | jq .job_id)
curl -s {{ B }}/jobs/$JOB -H "Authorization: Bearer $ZDRS_KEY" | jq .status
